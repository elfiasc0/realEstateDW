WITH 

-- Apartments with 2 buyers
MEMBER [Measures].[NumOfApt_2] AS
    Count(
        Filter(
            [Apartment].[ID Apartment].Members,
            [Measures].[Number of Potential Buyer] = 2
        )
    )

MEMBER [Measures].[TotalSaleSpeed_2] AS
    Sum(
        Filter(
            [Apartment].[ID Apartment].Members,
            [Measures].[Number of Potential Buyer] = 2
        ),
        [Measures].[AverageSaleSpeed]
    )

MEMBER [Measures].[AverageSaleSpeed_2] AS
    IIF(
        [Measures].[NumOfApt_2] = 0,
        NULL,
        [Measures].[TotalSaleSpeed_2] / [Measures].[NumOfApt_2]
    )
